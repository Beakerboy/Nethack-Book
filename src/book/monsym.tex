% Setup the pgfkeys lookup table structure
\pgfkeys{
  /monsym/.is family, /monsym/.unknown/.style={%
    /monsym/char/.initial=X,
    /monsym/color/.initial=white,
  }
}

% Helper macro to define monster properties (color, character)
\newcommand{\definemonster}[3]{%
  \pgfkeys{/monsym/#1/color/.initial=#2}%
  \pgfkeys{/monsym/#1/char/.initial=#3}%
}

% Define the \monsym command
\newcommand{\monsym}[1][]{%
  \colorbox{black}{%
    \textcolor{\pgfkeysvalueof{/monsym/#1/color}}{%
      \texttt{\pgfkeysvalueof{/monsym/#1/char}}%
    }%
  }%
}

\newcommand{\whitesym}[1][]{%
  \colorbox{black}{%
    \textcolor{white}{%
      \texttt{#1}%
    }%
  }%
}

\newcommand{\monsymlink}[1][]{%
  \monsym[#1]\mbox{$~$}\texttt{#1}%
  }
